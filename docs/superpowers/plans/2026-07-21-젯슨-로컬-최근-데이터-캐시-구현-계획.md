# 젯슨 로컬 최근 데이터 캐시 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Jetson `device-serial-jetson`이 6개 가상 디바이스의 최신값과 최근 10분 구간을 메모리에 보관하고 Workflow Pod에 Local Data API로 제공한다.

**Architecture:** Device Service driver 안에 `(deviceName, resourceName)`별 bounded recent cache를 두고 정상 파싱된 sample을 중앙 Event 발행 전에 저장한다. EdgeX Device SDK의 custom route로 latest/recent GET을 같은 `59910` 포트에 등록하며, 중앙 이력 권위는 Core Data/PostgreSQL로 유지한다.

**Tech Stack:** Go 1.25, EdgeX Device SDK v4.0.2, Echo v4, testify, Kubernetes/Kustomize, pytest, Argo CD

## Global Constraints

- cache는 각 Device/resource별 최근 10분, 최대 10,000 sample이며 Pod 재시작 시 초기화한다.
- recent 응답은 가장 최근 `limit`개를 선택한 뒤 `origin` 오름차순으로 반환한다.
- `limit` 기본값은 1,000이고 허용 범위는 1~10,000이다.
- latest sample 부재는 404, 알려진 source의 빈 recent는 200과 빈 `samples`다.
- 중앙 Core Data fallback, SQLite/PVC, MQTT broker, InfluxDB를 추가하지 않는다.
- 기존 Service FQDN을 사용하고 fixed ClusterIP/PodIP/node IP를 manifest나 client에 기록하지 않는다.
- EdgeMesh 구성을 변경하지 않는다.
- Workflow Pod ingress는 `edge-ai.io/local-data-client=true` label로 제한한다.

```go
func TestRecentCacheEvictsByAgeAndCapacity(t *testing.T) {
    cache := newRecentCache(10*time.Minute, 3)
    now := int64(20 * time.Minute)
    origins := []int64{now - int64(10*time.Minute) - 1, now - 3, now - 2, now - 1, now}
    for index, origin := range origins {
        cache.append("device", "resource", cachedSample{Origin: origin, ValueType: common.ValueTypeInt32, Value: int32(index)})
    }
    got := cache.query("device", "resource", now-int64(10*time.Minute), now, 10, now)
    require.Len(t, got, 3)
    assert.Equal(t, []int64{now - 2, now - 1, now}, []int64{got[0].Origin, got[1].Origin, got[2].Origin})
}

func TestRecentCacheReturnsNewestLimitInAscendingOrder(t *testing.T) {
    cache := newRecentCache(10*time.Minute, 10)
    now := int64(20 * time.Minute)
    for value := int32(1); value <= 5; value++ {
        cache.append("device", "resource", cachedSample{Origin: now - int64(6-value), ValueType: common.ValueTypeInt32, Value: value})
    }
    got := cache.query("device", "resource", now-10, now, 2, now)
    require.Len(t, got, 2)
    assert.Equal(t, []int32{4, 5}, []int32{got[0].Value, got[1].Value})
}

func TestRecentCacheDeleteDevice(t *testing.T) {
    cache := newRecentCache(10*time.Minute, 10)
    now := int64(20 * time.Minute)
    cache.append("remove", "resource", cachedSample{Origin: now, ValueType: common.ValueTypeInt32, Value: 1})
    cache.append("keep", "resource", cachedSample{Origin: now, ValueType: common.ValueTypeInt32, Value: 2})
    cache.deleteDevice("remove")
    _, removed := cache.latest("remove", "resource", now)
    kept, ok := cache.latest("keep", "resource", now)
    assert.False(t, removed)
    require.True(t, ok)
    assert.Equal(t, int32(2), kept.Value)
}

func TestRecentCacheConcurrentReadWrite(t *testing.T) {
    cache := newRecentCache(10*time.Minute, 100)
    now := int64(20 * time.Minute)
    var wg sync.WaitGroup
    for worker := 0; worker < 8; worker++ {
        wg.Add(1)
        go func(offset int) {
            defer wg.Done()
            for index := 0; index < 100; index++ {
                origin := now + int64(offset*100+index)
                cache.append("device", "resource", cachedSample{Origin: origin, ValueType: common.ValueTypeInt32, Value: int32(index)})
                _ = cache.query("device", "resource", now-int64(time.Minute), origin, 10, origin)
            }
        }(worker)
    }
    wg.Wait()
    assert.LessOrEqual(t, len(cache.query("device", "resource", now-int64(time.Minute), now+1_000, 10_000, now+1_000)), 100)
}
```

```go
func TestLocalDataAPIRecentReturnsEmptySamplesForKnownSource(t *testing.T) {
    api := newLocalDataAPI(newRecentCache(recentCacheMaxAge, recentCacheMaxSamples), func(device, resource string) bool { return true })
    api.now = func() int64 { return int64(20 * time.Minute) }
    recorder := httptest.NewRecorder()
    context := echo.New().NewContext(httptest.NewRequest(http.MethodGet, recentRoute, http.NoBody), recorder)
    context.SetParamNames("device", "resource")
    context.SetParamValues("virtual-temperature-001", "temperature_raw")
    require.NoError(t, api.recent(context))
    assert.Equal(t, http.StatusOK, recorder.Code)
    var response localDataResponse
    require.NoError(t, json.Unmarshal(recorder.Body.Bytes(), &response))
    assert.Empty(t, response.Samples)
    assert.Equal(t, 0, response.Count)
}

func TestLocalDataAPIReturnsNotFoundForUnknownSourceAndMissingLatest(t *testing.T) {
    tests := []struct {
        name  string
        known bool
    }{{name: "unknown source", known: false}, {name: "known source without latest", known: true}}
    for _, test := range tests {
        t.Run(test.name, func(t *testing.T) {
            api := newLocalDataAPI(newRecentCache(recentCacheMaxAge, recentCacheMaxSamples), func(device, resource string) bool { return test.known })
            recorder := httptest.NewRecorder()
            context := echo.New().NewContext(httptest.NewRequest(http.MethodGet, latestRoute, http.NoBody), recorder)
            context.SetParamNames("device", "resource")
            context.SetParamValues("virtual-temperature-001", "temperature_raw")
            require.NoError(t, api.latest(context))
            assert.Equal(t, http.StatusNotFound, recorder.Code)
        })
    }
}

func TestLocalDataAPIRejectsInvalidQuery(t *testing.T) {
    for _, rawQuery := range []string{"from=nope", "from=2&to=1", "limit=0", "limit=10001"} {
        t.Run(rawQuery, func(t *testing.T) {
            api := newLocalDataAPI(newRecentCache(recentCacheMaxAge, recentCacheMaxSamples), func(device, resource string) bool { return true })
            request := httptest.NewRequest(http.MethodGet, recentRoute+"?"+rawQuery, http.NoBody)
            recorder := httptest.NewRecorder()
            context := echo.New().NewContext(request, recorder)
            context.SetParamNames("device", "resource")
            context.SetParamValues("virtual-temperature-001", "temperature_raw")
            require.NoError(t, api.recent(context))
            assert.Equal(t, http.StatusBadRequest, recorder.Code)
        })
    }
}
```
