# EdgeX 디바이스 조회 단일 실행 구현 계획

> 상태: 설계 이력. 현재 운영 기능의 완료 근거가 아니며 구현 계획과 검증 맥락으로 보존한다.

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Prevent concurrent state-aggregator endpoints from issuing duplicate EdgeX device observations that exhaust the Core Data PostgreSQL connection pool.

**Architecture:** `StateAggregatorService.get_devices()` shares only its currently running observation task. The completed result is never cached; success and failure both clear the active task so the next request performs a fresh EdgeX read.

**Tech Stack:** Python 3.11, asyncio, FastAPI, pytest

## Global Constraints

- Physical device inventory, state, and telemetry authority remains EdgeX.
- Do not cache the last successful device observation or present stale values as current.
- Do not turn an EdgeX backend error into a successful empty inventory response.
- Do not mutate EdgeX metadata/state, publish commands, or execute workflows.
- Preserve unrelated working-tree changes and stage only the named files.

```python
def test_device_observation_does_not_cache_completed_result(monkeypatch, tmp_path):
    aggregator = StateAggregatorService(Settings(data_dir=tmp_path))
    calls = 0

    async def fake_devices():
        nonlocal calls
        calls += 1
        return [_edgex_device("sensor-01")]

    async def fake_latest(device_name: str):
        return [_reading(device_name, datetime.now(timezone.utc))]

    monkeypatch.setattr(aggregator.edgex, "get_devices", fake_devices)
    monkeypatch.setattr(aggregator.edgex, "get_latest_source_readings", fake_latest)

    async def scenario():
        first = await aggregator.get_devices()
        second = await aggregator.get_devices()
        return first, second

    first, second = asyncio.run(scenario())

    assert calls == 2
    assert first[0].name == second[0].name == "sensor-01"


def test_device_observation_failure_is_shared_and_next_call_retries(monkeypatch, tmp_path):
    aggregator = StateAggregatorService(Settings(data_dir=tmp_path))
    calls = 0

    async def scenario():
        nonlocal calls
        started = asyncio.Event()
        release = asyncio.Event()

        async def fake_devices():
            nonlocal calls
            calls += 1
            if calls == 1:
                started.set()
                await release.wait()
                raise EdgeXBackendError("Core Data unavailable")
            return [_edgex_device("sensor-01")]

        async def fake_latest(device_name: str):
            return [_reading(device_name, datetime.now(timezone.utc))]

        monkeypatch.setattr(aggregator.edgex, "get_devices", fake_devices)
        monkeypatch.setattr(aggregator.edgex, "get_latest_source_readings", fake_latest)

        failed = [asyncio.create_task(aggregator.get_devices()) for _ in range(2)]
        await started.wait()
        await asyncio.sleep(0)
        release.set()
        failures = await asyncio.gather(*failed, return_exceptions=True)
        recovered = await aggregator.get_devices()
        return failures, recovered

    failures, recovered = asyncio.run(scenario())

    assert calls == 2
    assert all(isinstance(error, EdgeXBackendError) for error in failures)
    assert recovered[0].name == "sensor-01"
```

Replace `get_devices` with:

```python
async def get_devices(self) -> list[DeviceState]:
    task = self._device_observation_task
    if task is None or task.done():
        task = asyncio.create_task(self._fetch_devices())
        self._device_observation_task = task
    try:
        return list(await asyncio.shield(task))
    finally:
        if task.done() and self._device_observation_task is task:
            self._device_observation_task = None

async def _fetch_devices(self) -> list[DeviceState]:
    inventory = await self.edgex.get_devices()
    latest_events = await asyncio.gather(
        *(self.edgex.get_latest_source_readings(device.name) for device in inventory)
    )
    return [
        self._normalize_edgex_device(device, readings)
        for device, readings in zip(inventory, latest_events)
    ]
```
