For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Jetson의 실제 Arduino 가속도 X/Y/Z Local Data를 준실시간으로 결합·분석하고, 온라인 이상 탐지 상태와 Device→Device Service→AI consumer 관계를 중앙 dashboard에 표시한다.

Architecture: 새 sensor-anomaly-demo FastAPI 서비스가 device-serial-jetson Local Data API를 origin cursor로 읽고 세 축 frame을 결합한다. Jetson에서 온라인 Gaussian baseline을 실행하며, state-aggregator는 read-only Service FQDN으로 결과를 정규화해 dashboard에 표시한다. 원시 telemetry/history 권위는 중앙 EdgeX Core Data/PostgreSQL에 유지한다.

Tech Stack: Python 3.11, FastAPI 0.116.1, httpx 0.28.1, Pydantic 2.11.3, pytest 8.4.2, vanilla JavaScript, Kubernetes/Kustomize, Argo CD, Docker buildx

Global Constraints

  • 모든 shell 명령은 저장소 규칙에 따라 rtk로 시작한다.
  • 구현은 테스트를 먼저 실패시킨 뒤 최소 구현으로 통과시키는 TDD 순서를 따른다.
  • runtime 데이터 경로는 Service FQDN만 사용하며 fixed ClusterIP, PodIP 또는 node IP를 사용하지 않는다.
  • sensor-anomaly-demo는 etri-dev0001-jetorn에 배치하고 edge-ai.io/local-data-client=true label을 가진다.
  • Local Data polling 기본 간격은 0.5초, input stale 기준은 10초다.
  • 세 축 Int32 sample은 같은 nanosecond origin일 때만 하나의 frame으로 처리한다.
  • warm-up 30 frame, threshold 4.0, 2회 연속 초과 시 anomaly, 3회 연속 정상 시 clear, EWMA alpha 0.05를 사용한다.
  • API는 read-only이며 합성 anomaly 주입, actuator command, EdgeX metadata/state mutation을 추가하지 않는다.
  • local MQTT broker, InfluxDB, Agent, SQLite/PVC, hostNetwork를 추가하지 않는다.
  • EdgeMesh manifest, image, ConfigMap과 DaemonSet을 수정하지 않는다.
  • 기존 vision_stage_runner의 가짜 결과는 운영 데모 근거로 사용하지 않고 해당 Argo Application을 퇴역한다.
  • 현재 Flannel이 NetworkPolicy를 집행하지 않는다는 한계를 문서와 운영 결과에 유지한다.

# app/models.py
from dataclasses import dataclass
from typing import Literal

AxisName = Literal["x", "y", "z"]

@dataclass(frozen=True)
class AxisSample:
    origin: int
    value_type: str
    value: int

@dataclass(frozen=True)
class AccelerationFrame:
    origin: int
    x: int
    y: int
    z: int

@dataclass
class JoinCounters:
    duplicates_ignored: int = 0
    incomplete_frames_dropped: int = 0
# app/joiner.py 핵심 계약
class FrameJoiner:
    def __init__(self, pending_ttl_ns: int, max_pending: int) -> None:
        if pending_ttl_ns <= 0 or max_pending <= 0:
            raise ValueError("pending limits must be positive")
        self.pending_ttl_ns = pending_ttl_ns
        self.max_pending = max_pending
        self.counters = JoinCounters()
        self._pending: dict[int, dict[AxisName, int]] = {}
        self._pending_seen_at: dict[int, int] = {}
        self._processed_seen_at: dict[int, int] = {}

    def ingest(
        self,
        axis: AxisName,
        samples: list[AxisSample],
        now_ns: int,
    ) -> list[AccelerationFrame]:
        self._evict(now_ns)
        frames: list[AccelerationFrame] = []
        for sample in samples:
            if sample.origin in self._processed_seen_at:
                self.counters.duplicates_ignored += 1
                continue
            values = self._pending.setdefault(sample.origin, {})
            self._pending_seen_at.setdefault(sample.origin, now_ns)
            if axis in values:
                self.counters.duplicates_ignored += 1
            values[axis] = sample.value
            if set(values) == {"x", "y", "z"}:
                frames.append(AccelerationFrame(
                    origin=sample.origin,
                    x=values["x"], y=values["y"], z=values["z"],
                ))
                self._pending.pop(sample.origin)
                self._pending_seen_at.pop(sample.origin)
                self._processed_seen_at[sample.origin] = now_ns
        self._evict(now_ns)
        return sorted(frames, key=lambda item: item.origin)

    @property
    def pending_origins(self) -> list[int]:
        return sorted(self._pending)

    def _evict(self, now_ns: int) -> None:
        cutoff = now_ns - self.pending_ttl_ns
        expired = [origin for origin, seen in self._pending_seen_at.items() if seen < cutoff]
        capacity = max(0, len(self._pending) - len(expired) - self.max_pending)
        survivors = [origin for origin in sorted(self._pending) if origin not in set(expired)]
        for origin in expired + survivors[:capacity]:
            self._pending.pop(origin, None)
            self._pending_seen_at.pop(origin, None)
            self.counters.incomplete_frames_dropped += 1
        for origin, seen in list(self._processed_seen_at.items()):
            if seen < cutoff:
                self._processed_seen_at.pop(origin)

구현은 dict[int, dict[AxisName, int]] pending map과 processed-origin set을 사용한다. 완전한 frame은 origin 오름차순으로 반환하고 processed origin은 다시 방출하지 않는다. TTL과 capacity eviction 모두 incomplete_frames_dropped를 정확히 한 번 올린다.

class OnlineGaussianDetector:
    def process(self, frame: AccelerationFrame) -> InferenceResult:
        magnitude = math.sqrt(frame.x**2 + frame.y**2 + frame.z**2)
        if self._count < self.config.warmup_samples:
            self._count += 1
            delta = magnitude - self._mean
            self._mean += delta / self._count
            self._m2 += delta * (magnitude - self._mean)
            if self._count == self.config.warmup_samples:
                self._variance = self._m2 / max(1, self._count - 1)
                status = "normal"
            else:
                status = "warming_up"
            return self._result(frame, magnitude, 0.0, status)

        stddev = max(math.sqrt(max(self._variance, 0.0)), self.config.stddev_floor)
        score = abs(magnitude - self._mean) / stddev
        exceeds = score >= self.config.threshold
        if exceeds:
            self._above += 1
            self._below = 0
            if self._above >= self.config.anomaly_streak:
                self._anomaly = True
        else:
            self._below += 1
            self._above = 0
            if self._anomaly and self._below >= self.config.clear_streak:
                self._anomaly = False
            alpha = self.config.ewma_alpha
            delta = magnitude - self._mean
            self._mean += alpha * delta
            self._variance = (1 - alpha) * (self._variance + alpha * delta * delta)
            self._count += 1
        status = "anomaly" if self._anomaly else "normal"
        return self._result(frame, magnitude, score, status)

InferenceResult는 origin, X/Y/Z, magnitude, score, anomaly, status를 갖고, status는 warming_up | normal | anomaly로 제한한다. ModelSnapshot은 algorithm, sample_count, warmup_samples, threshold, baseline_mean/stddev를 반환한다. float API 값은 소수점 6자리에서 반올림해 동일 입력에 안정적인 JSON을 만든다.

class AnomalyRuntime:
    async def start(self) -> None:
        if self._task is None:
            self._stop.clear()
            self._task = asyncio.create_task(self._run())

    async def stop(self) -> None:
        self._stop.set()
        if self._task is not None:
            await self._task
            self._task = None

    async def poll_once(self, now_ns: int | None = None) -> None:
        current = time.time_ns() if now_ns is None else now_ns
        responses = await asyncio.gather(
            *(self._fetch_source(source, current) for source in self.sources),
            return_exceptions=True,
        )
        for source, response in zip(self.sources, responses, strict=True):
            if isinstance(response, Exception):
                self._record_input_error(source, response)
                continue
            frames = self.joiner.ingest(source.axis, response, current)
            for frame in frames:
                result = self.detector.process(frame)
                self._results.append(result)
                self._last_origin = result.origin
                self._frames_processed += 1

    def status(self, now_ns: int | None = None) -> ServiceStatus:
        current = time.time_ns() if now_ns is None else now_ns
        return self._build_status(current)

    def results(self, limit: int) -> list[InferenceResult]:
        if limit < 1 or limit > self._results.maxlen:
            raise ValueError("limit outside result retention")
        return list(self._results)[-limit:]

세 fetch는 asyncio.gather(*requests, return_exceptions=True)와 같은 실패 격리 방식으로 실행한다. source별 cursor는 성공 응답의 최대 origin으로만 전진한다. 일부 source 오류여도 성공 sample은 pending joiner에 전달하지만 완전 frame이 아니면 inference하지 않는다. worker loop는 0.5초 주기를 Event.wait() timeout으로 구현해 shutdown이 즉시 끝나게 한다. 최근 결과는 deque(maxlen=1000)다.
